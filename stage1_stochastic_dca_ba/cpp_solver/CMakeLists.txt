cmake_minimum_required(VERSION 3.16)
project(stochastic_dca LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Eigen3 3.4 QUIET NO_MODULE)
add_executable(stochastic_dca stochastic_dca.cpp)
if(Eigen3_FOUND)
  target_link_libraries(stochastic_dca PRIVATE Eigen3::Eigen)
elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/eigen/Eigen/Dense")
  target_include_directories(stochastic_dca PRIVATE third_party/eigen)
else()
  message(FATAL_ERROR "Eigen3 not found; use the bundled third_party/eigen headers")
endif()
target_compile_definitions(stochastic_dca PRIVATE EIGEN_DONT_PARALLELIZE EIGEN_NO_DEBUG)
if(MSVC)
  target_compile_options(stochastic_dca PRIVATE /O2 /fp:precise)
else()
  target_compile_options(stochastic_dca PRIVATE -O3 -march=native)
endif()
